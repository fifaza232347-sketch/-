async function sendCmd(cmd) { if (!writeChar) { document.getElementById("status").innerText = "กรุณากดเชื่อมต่อ micro:bit ก่อน"; return; } if (cmd !== "S" && isSending) { pendingCmd = cmd; return; } isSending = true; const data = new TextEncoder().encode(cmd + "\n"); try { if (writeChar.properties.writeWithoutResponse) { await writeChar.writeValueWithoutResponse(data); } else if (writeChar.properties.write) { await writeChar.writeValue(data); } document.getElementById("status").innerText = "ส่งคำสั่งแล้ว: " + cmd; } catch (err) { document.getElementById("status").innerText = "ส่งคำสั่งไม่สำเร็จ: " + err.message; } finally { isSending = false; if (pendingCmd) { const next = pendingCmd; pendingCmd = null; sendCmd(next); } } }
